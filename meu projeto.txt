    function formatPressure(raw) {
      if (state.settings.pressureUnit !== 'psi') return raw;
      return raw.replace(/(\d+(?:\.\d+)?)\s*bar/gi, (_, num) => `${(parseFloat(num) * 14.5038).toFixed(1)} PSI`);

    function buildDiagnosticReport(rawCode) {
      const clean = rawCode.trim().toUpperCase();
      const chosenCategory = state.selectedCategory || 'Trator Agrícola';
      const chosenBrand = state.selectedBrand || VEHICLE_CATALOG[chosenCategory].brands[0];
      const voltage = VEHICLE_CATALOG[chosenCategory]?.voltage || '12V / 24V';

      const exact = FAULT_DATABASE.find(item => item.code.toUpperCase() === clean);
      if (exact) {
        return {
          ...exact,
          category: state.selectedCategory || exact.category,
          brand: state.selectedBrand || exact.brand,
          voltage: VEHICLE_CATALOG[state.selectedCategory || exact.category]?.voltage || exact.voltage
        };
      }

      const isHeavyOrAgri = ['Caminhão', 'Trator Agrícola', 'Colheitadeira', 'Linha Amarela'].includes(chosenCategory);
      const isFuelOrBoost = /0087|0299|0234|0094|157|179|RAIL|TURBO|PRESS|FUEL|COMBUST/.test(clean);
      const isElectricalOrNet = /^U|^B|0100|0121|0562|0247|FMI\s*0?[345]|CAN|ECU|VOLT/.test(clean);

      if (isElectricalOrNet) {
        return {
          code: clean,
          category: chosenCategory,
          brand: chosenBrand,
          system: isHeavyOrAgri ? 'Rede SAE J1939 & Alimentação' : 'Rede CAN & Alimentação',
          severity: 'ALTA',
          voltage,
          title: `Anomalia de sinal, alimentação ou comunicação (${clean})`,
          meaning: `Módulo de controle em ${chosenBrand} (${chosenCategory}) registrou variação de tensão, resistência ou perda de quadro no código ${clean}.`,
          causes: [
            { label: `Queda de tensão, mau contato em conector ou oxidação em chicote (${voltage})`, prob: 45 },
            { label: 'Curto-circuito para massa ou +B em sensor/atuador associado', prob: 35 },
            { label: 'Interferência em barramento CAN ou falha interna de módulo', prob: 20 }
          ],
          readings: {
            volts: isHeavyOrAgri ? '24.0V linha • 5.0V ref sensor' : '12.6V linha • 5.0V ref sensor',
            ohms: '< 1.0 Ω continuidade / ~60 Ω barramento CAN',
            pressure: 'N/A (Sinal / Rede)'
          },
          checklist: [
            `Medir tensão de alimentação (${voltage}) sob carga e verificar queda de massa.`,
            'Inspecionar pinos e travas do conector mais próximo ao circuito afetado.',
            'Testar continuidade e isolamento contra massa/positivo com multímetro.'
          ]
        };
      }

      return {
        code: clean,
        category: chosenCategory,
        brand: chosenBrand,
        system: isHeavyOrAgri ? 'Trem de Força & Sensores' : 'Motor & Gerenciamento',
        severity: isFuelOrBoost ? 'CRÍTICA' : 'ALTA',
        voltage,
        title: `Falha registrada no código ${clean} (${chosenBrand})`,
        meaning: `Diagnóstico direcionado para ${chosenBrand} na categoria ${chosenCategory} (${voltage}) referente ao código ${clean}.`,

    async function fetchGeminiWithBackoff(userText, historySnapshot) {
      const apiKey = "";
      const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

      const contents = historySnapshot.slice(-8).map(item => ({
        role: item.role === 'user' ? 'user' : 'model',
        parts: [{ text: item.text }]
      }));
      contents.push({ role: 'user', parts: [{ text: userText }] });

      const payload = {
        contents,
        systemInstruction: {
          parts: [{ text: buildAssistantSystemPrompt() }]
        }
      };

      const delays = [1000, 2000];
      for (let attempt = 0; attempt <= delays.length; attempt++) {
        try {
          const resp = await fetch(apiUrl, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
          });
          if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
          const data = await resp.json();
          const replyText = data?.candidates?.[0]?.content?.parts?.[0]?.text;
          if (replyText) return replyText.trim();
        } catch (_) {
          if (attempt < delays.length) {
            await new Promise(r => setTimeout(r, delays[attempt]));
          }
        }
      }
      return null;
    }

    function buildLocalVersatileReply(query) {
      const user = state.activeUser || { name: 'Operador', age: 27 };

      const asksName = /nome|name|nombre|prénom|nom|名前|이름|имя|اسم/.test(q);
      const asksAge = /idade|anos|age|edad|âge|ans|alter|età|年齢|나이|возраст|عمر/.test(q);

      if (asksName && asksAge) {
        if (lang === 'en') return `Your name is **${user.name}** and you are **${user.age} years old**.`;
        if (lang === 'es') return `Tu nombre es **${user.name}** y tienes **${user.age} años**.`;
        return `Seu nome é **${user.name}** e você tem **${user.age} anos**.`;
      }
      if (asksName) {
        if (lang === 'en') return `Your name is **${user.name}**.`;
        if (lang === 'es') return `Tu nombre es **${user.name}**.`;
        return `Seu nome é **${user.name}**.`;
      }
      if (asksAge) {
        if (lang === 'en') return `You are **${user.age} years old**, ${user.name}.`;
        if (lang === 'es') return `Tienes **${user.age} años**, ${user.name}.`;
        return `Você tem **${user.age} anos**, ${user.name}.`;
      }

      if (state.chatMode === 'electrical') {
        return `Entendido, **${user.name}** (modo **Elétrica & Pinagem**): para **"${query}"**, verifique primeiro alimentação nominal (12V/24V), referência de **5.0V**, aterramento (< 0.1V de queda) e resistência/sinal do circuito com multímetro ou osciloscópio.`;

        return `Certo, **${user.name}** (modo **Bancada & Medições**): sobre **"${query}"**, isole o componente da máquina, confira resistência em **Ohms (Ω)**, tensão em **Volts (V)** e pressão em **${state.settings.pressureUnit.toUpperCase()}** sob simulação controlada.`;
      }
      return `Entendido, **${user.name}**! Sobre **"${query}"**: vamos direto ao ponto — comece pelos fundamentos principais, valide os dados ou condições reais e siga por etapas. Quer que eu detalhe algum ponto específico?`;
    }

    async function sendChatMessage() {
      const input = document.getElementById('chat-text-input');
      const userText = input.value.trim();
      if (!userText) return;

      input.value = '';
      const historySnapshot = [...state.chatHistory];
      state.chatHistory.push({ role: 'user', text: userText });

      const pendingIdx = state.chatHistory.push({
        role: 'ai',
        text: '...',
        relatedCodes: []
      }) - 1;
      renderChatMessages();

      const aiText = (await fetchGeminiWithBackoff(userText, historySnapshot)) || buildLocalVersatileReply(userText);
      const codeMatch = userText.toUpperCase().match(/\b([PUBC][0-9]{4}|SPN\s?[0-9]+|ECU\s?[0-9.]+|CID\s?[0-9]+|[0-9]{5})\b/);

      state.chatHistory[pendingIdx] = {
        role: 'ai',
        text: aiText,
        relatedCodes: codeMatch ? [codeMatch[0]] : []
      };
      renderChatMessages();

      if (state.settings.autoSpeak) {
        speakPlainText(aiText.replace(/\*\*/g, ''));
      }
    }

          const bg = prio === 'Crítica' ? 'bg-neonRed' : prio === 'Alta' ? 'bg-neonAmber' : 'bg-neonCyan';
          btn.classList.add(bg, 'text-black', 'font-extrabold');
        } else {
          btn.classList.add('text-slate-400', 'font-bold');
        }
      });
    }

    function setFixFilter(filter) {
      state.fixStatusFilter = filter;
      document.querySelectorAll('.fix-filter-pill').forEach(pill => {
        const active = pill.dataset.ffilter === filter;
        pill.classList.toggle('bg-neonCyan', active);
        pill.classList.toggle('text-black', active);
        pill.classList.toggle('font-extrabold', active);
        pill.classList.toggle('bg-surfaceLight', !active);
        pill.classList.toggle('text-slate-300', !active);
        pill.classList.toggle('font-bold', !active);
      });
      renderFixTickets();
    }

    function submitTechnicalFix() {
      const module = document.getElementById('fix-module').value;
      const title = document.getElementById('fix-title').value.trim();
      const desc = document.getElementById('fix-desc').value.trim();

      if (!title || title.length < 4) {
        showNativeToast('Escreva o título do bug ou correção');
        document.getElementById('fix-title').focus();
        return;
      }

      const nextNum = 101 + state.fixTickets.length;
      const newTicket = {
        id: `TCK-${nextNum}`,
        module,
        priority: state.selectedFixPriority,
        status: 'Aberto',
        title,
        desc: desc || 'Correção técnica registrada pelo operador no terminal.',
        author: state.activeUser?.name || 'Operador',
        date: 'Agora'
      };

      state.fixTickets = [newTicket, ...state.fixTickets];
      localStorage.setItem('diagcore_fixes_v1', JSON.stringify(state.fixTickets));

      document.getElementById('fix-title').value = '';
      document.getElementById('fix-desc').value = '';
      renderFixTickets();
      showNativeToast(`${newTicket.id} registrado!`);
    }

    function cycleTicketStatus(id) {
      const order = ['Aberto', 'Em Análise', 'Corrigido'];
      state.fixTickets = state.fixTickets.map(t => {
        if (t.id !== id) return t;
        const nextStatus = order[(order.indexOf(t.status) + 1) % order.length];
        showNativeToast(`${t.id}: ${nextStatus}`);
        return { ...t, status: nextStatus };
      });
      localStorage.setItem('diagcore_fixes_v1', JSON.stringify(state.fixTickets));
      renderFixTickets();
    }

    function deleteTicket(id) {
      state.fixTickets = state.fixTickets.filter(t => t.id !== id);
      localStorage.setItem('diagcore_fixes_v1', JSON.stringify(state.fixTickets));
      renderFixTickets();
      showNativeToast('Registro removido');
    }

    function renderFixTickets() {
      const container = document.getElementById('fixes-tickets-list');
      const openCount = state.fixTickets.filter(t => t.status !== 'Corrigido').length;
      document.getElementById('fixes-open-badge').textContent = `${openCount} ${openCount === 1 ? 'aberto' : 'abertos'}`;

      const filtered = state.fixStatusFilter === 'ALL'
        ? state.fixTickets
        : state.fixTickets.filter(t => t.status === state.fixStatusFilter);

      if (filtered.length === 0) {
        container.innerHTML = `
          <div class="p-6 rounded-3xl bg-surface border border-white/10 text-center">
            <p class="text-xs font-bold text-slate-300">Nenhum registro neste filtro</p>
            <p class="text-[11px] text-slate-500 mt-1">Use o formulário acima para registrar um bug ou correção.</p>
          </div>
        `;
        return;
      }

      container.innerHTML = filtered.map(t => {
        const statusColors = {
          'Aberto': 'bg-neonRed/15 text-neonRed border-neonRed/40',
          'Em Análise': 'bg-neonAmber/15 text-neonAmber border-neonAmber/40',
          'Corrigido': 'bg-neonGreen/15 text-neonGreen border-neonGreen/40'
        };
        const prioColors = {
          'Crítica': 'text-neonRed',
          'Alta': 'text-neonAmber',
          'Normal': 'text-neonCyan'
        };

        return `
          <div class="p-3.5 rounded-3xl bg-surface border border-white/10 space-y-2">
            <div class="flex items-center justify-between gap-2">
              <div class="flex items-center gap-1.5 flex-wrap">
                <span class="font-mono text-[10px] font-extrabold px-2 py-0.5 rounded bg-white/10 text-white">${t.id}</span>
                <span class="text-[10px] font-bold ${prioColors[t.priority] || 'text-neonCyan'}">● ${t.priority}</span>
                <span class="text-[10px] text-slate-400">• ${t.module}</span>
              </div>
              <button onclick="deleteTicket('${t.id}')" class="text-slate-500 hover:text-neonRed text-xs px-1" title="Excluir">✕</button>
            </div>

            <h4 class="text-xs font-black text-white leading-snug">${t.title}</h4>
            <p class="text-[11px] text-slate-300 leading-relaxed">${t.desc}</p>

            <div class="pt-1.5 border-t border-white/[0.06] flex items-center justify-between">
              <span class="text-[10px] font-mono text-slate-500">Por: ${t.author}</span>
              <button onclick="cycleTicketStatus('${t.id}')" class="px-2.5 py-1 rounded-full border text-[10px] font-extrabold active:scale-95 transition ${statusColors[t.status] || statusColors['Aberto']}">
                Status: ${t.status} ↻
              </button>
            </div>
          </div>
        `;
      }).join('');
    }

    function toggleSaveCode(dtc) {
      const exists = isCodeSaved(dtc.code, dtc.brand);
      if (exists) {
        state.savedCodes = state.savedCodes.filter(item => !(item.code === dtc.code && item.brand === dtc.brand));
        showNativeToast(`${dtc.code} removido`);
      } else {
        state.savedCodes = [dtc, ...state.savedCodes];
        showNativeToast(`${dtc.code} salvo!`);
      }

      localStorage.setItem('diagcore_saved_v6', JSON.stringify(state.savedCodes));
      updateSavedBadge();
      renderFeed();
      if (!document.getElementById('view-result-overlay').classList.contains('hidden')) {
        openResultOverlay(dtc);
      }
    }

    function updateSavedBadge() {
      const dot = document.getElementById('saved-count-dot');
      if (state.savedCodes.length > 0) {
        dot.textContent = state.savedCodes.length;
        dot.classList.remove('hidden');
      } else {
        dot.classList.add('hidden');
      }
    }

    function openSavedDrawer() {
      renderSavedList();
      document.getElementById('saved-drawer-modal').classList.remove('hidden');
      lucide.createIcons();
    }

    function closeSavedDrawer() {
      document.getElementById('saved-drawer-modal').classList.add('hidden');
    }

    function renderSavedList() {
      const container = document.getElementById('saved-list-container');
      if (state.savedCodes.length === 0) {
        container.innerHTML = `
          <div class="py-12 text-center">
            <p class="text-sm font-bold text-slate-300">Nenhum código salvo ainda</p>
            <p class="text-xs text-slate-500 mt-1">Toque em Salvar em qualquer laudo para guardar aqui.</p>
          </div>
        `;
        return;
      }

      container.innerHTML = state.savedCodes.map(dtc => {
        const safeJson = JSON.stringify(dtc).replace(/'/g, "&#39;");
        const catIcon = VEHICLE_CATALOG[dtc.category]?.icon || '⚙️';
        return `
          <button onclick='closeSavedDrawer(); openResultOverlay(${safeJson})' class="w-full text-left p-3.5 rounded-3xl bg-surfaceLight border border-white/10 transition flex items-center justify-between gap-3 active:scale-[0.98]">
            <div class="min-w-0">
              <div class="flex items-center gap-1.5 flex-wrap">
                <span class="font-mono font-black text-sm text-neonCyan">${dtc.code}</span>
                <span class="text-[10px] font-bold px-2 py-0.5 rounded-full bg-neonAmber/20 text-neonAmber">${dtc.brand}</span>
                <span class="text-[10px] font-semibold px-2 py-0.5 rounded-full bg-white/10 text-slate-300">${catIcon} ${dtc.category}</span>
              </div>
              <div class="text-xs font-bold text-white truncate mt-1">${dtc.title}</div>
            </div>
            <i data-lucide="chevron-right" class="w-4 h-4 text-slate-400 shrink-0"></i>
          </button>
        `;
      }).join('');
    }

    function clearSavedCodes() {
      state.savedCodes = [];
      localStorage.removeItem('diagcore_saved_v6');
      updateSavedBadge();
      renderSavedList();
      renderFeed();
      showNativeToast('Salvos limpos');
    }

    function speakDiagnostic(dtc) {
      const text = `Código ${dtc.code}, ${dtc.brand}. ${dtc.title}. ${dtc.meaning}. Teste: ${dtc.checklist?.[0] || ''}`;
      speakPlainText(text);
    }

    function speakPlainText(text) {
      if (!('speechSynthesis' in window)) {
        showNativeToast('Voz indisponível neste navegador');
        return;
      }
      if (window.speechSynthesis.speaking) {
        window.speechSynthesis.cancel();
        return;
      }
      const langMeta = SUPPORTED_LANGUAGES.find(l => l.code === state.settings.language) || SUPPORTED_LANGUAGES[0];
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.lang = langMeta.tts;
      utterance.rate = 1.05;
      window.speechSynthesis.speak(utterance);
      showNativeToast(`🔊 ${langMeta.badge}`);
    }

    function stopSpeech() {
      if ('speechSynthesis' in window && window.speechSynthesis.speaking) {
        window.speechSynthesis.cancel();
      }
    }
